cmake_minimum_required(VERSION 3.24)
project(jshc_server_driver LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "本入口只验收Linux服务器；不把其他系统结果标为Linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(jshc_server_driver SHARED driver.cpp)
target_link_libraries(jshc_server_driver PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(jshc_server_driver PRIVATE -O3 -DNDEBUG -Wall -Wextra -fvisibility=hidden)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/driver-$<CONFIG>.json"
  CONTENT "{\"path\":\"$<TARGET_FILE:jshc_server_driver>\"}\n")

cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
project(jshc_gpu LANGUAGES CXX CUDA)
add_executable(jshc_gpu main.cu)
target_compile_features(jshc_gpu PRIVATE cxx_std_17 cuda_std_17)
target_include_directories(jshc_gpu PRIVATE ../cpp/third_party)
set_target_properties(jshc_gpu PROPERTIES CUDA_RUNTIME_LIBRARY Static)

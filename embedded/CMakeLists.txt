cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(jshc_embedded LANGUAGES C)
  include(CTest)
else()
  enable_language(C)
endif()

add_library(jshc_embedded STATIC src/jshc_embedded.c)
target_include_directories(jshc_embedded PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(jshc_embedded PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jshc_embedded PRIVATE -Wall -Wextra -Werror -pedantic -ffreestanding -fno-builtin)
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(jshc_embedded PRIVATE -fstack-usage)
  endif()
endif()

option(JSHC_EMBEDDED_FREERTOS "构建 Cortex-M3 FreeRTOS/QEMU 契约测试固件" OFF)
if(JSHC_EMBEDDED_FREERTOS)
  add_subdirectory(freertos)
elseif(BUILD_TESTING)
  add_executable(jshc_embedded_tests tests/contract.c tests/host.c)
  target_link_libraries(jshc_embedded_tests PRIVATE jshc_embedded)
  add_test(NAME embedded_contract COMMAND jshc_embedded_tests)
endif()

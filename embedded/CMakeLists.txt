cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(jshc_embedded LANGUAGES C)
  include(CTest)
else()
  enable_language(C)
endif()

set(JSHC_PORTABLE_PARITY_DEFAULT OFF)
if(TARGET jshc_core AND JSHC_ENABLE_PORTABLE_C)
  set(JSHC_PORTABLE_PARITY_DEFAULT ON)
endif()
option(JSHC_EMBEDDED_PORTABLE_PARITY "在相同夹具中核对 portable/embedded 和独立整数参考"
  ${JSHC_PORTABLE_PARITY_DEFAULT})
if(JSHC_EMBEDDED_PORTABLE_PARITY)
  add_library(jshc_portable_contract STATIC ../portable/src/jshc_portable.c)
  target_include_directories(jshc_portable_contract PUBLIC ../portable/include)
  target_compile_definitions(jshc_portable_contract INTERFACE JSHC_EMBEDDED_PORTABLE_PARITY=1)
  set_target_properties(jshc_portable_contract PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(jshc_portable_contract PRIVATE -Wall -Wextra -Werror -pedantic
      -ffreestanding -fno-builtin -fstack-usage)
  endif()
endif()
function(jshc_attach_portable_contract target)
  if(JSHC_EMBEDDED_PORTABLE_PARITY)
    target_link_libraries(${target} PRIVATE jshc_portable_contract)
  endif()
endfunction()

option(JSHC_EMBEDDED_SINGLE_HEADER_PARITY "在宿主及三个QEMU目标核对新C99单头与独立整数参考" ON)
if(JSHC_EMBEDDED_SINGLE_HEADER_PARITY)
  add_library(jshc_single_header_contract STATIC tests/single_header_runtime.c)
  target_include_directories(jshc_single_header_contract PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
  target_compile_definitions(jshc_single_header_contract INTERFACE JSHC_EMBEDDED_SINGLE_HEADER_PARITY=1)
  set_target_properties(jshc_single_header_contract PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(jshc_single_header_contract PRIVATE -Wall -Wextra -Werror -pedantic
      -ffreestanding -fno-builtin -fstack-usage)
  endif()
endif()
function(jshc_attach_single_header_contract target)
  if(JSHC_EMBEDDED_SINGLE_HEADER_PARITY)
    target_link_libraries(${target} PRIVATE jshc_single_header_contract)
  endif()
endfunction()

add_library(jshc_embedded STATIC src/jshc_embedded.c)
target_include_directories(jshc_embedded PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(jshc_embedded PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jshc_embedded PRIVATE -Wall -Wextra -Werror -pedantic -ffreestanding -fno-builtin)
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(jshc_embedded PRIVATE -fstack-usage)
  endif()
endif()

option(JSHC_EMBEDDED_FREERTOS "构建 Cortex-M3 或 RV32 FreeRTOS/QEMU 契约测试固件" OFF)
option(JSHC_EMBEDDED_LINUX_QEMU "构建 Cortex-A53 Linux/QEMU 测试 init" OFF)
if(JSHC_EMBEDDED_FREERTOS AND JSHC_EMBEDDED_LINUX_QEMU)
  message(FATAL_ERROR "FreeRTOS 与 Linux 必须使用不同构建目录")
endif()
if(JSHC_EMBEDDED_FREERTOS)
  add_subdirectory(freertos)
elseif(JSHC_EMBEDDED_LINUX_QEMU)
  add_subdirectory(linux)
elseif(BUILD_TESTING)
  add_executable(jshc_embedded_tests tests/contract.c tests/host.c)
  target_link_libraries(jshc_embedded_tests PRIVATE jshc_embedded)
  jshc_attach_portable_contract(jshc_embedded_tests)
  jshc_attach_single_header_contract(jshc_embedded_tests)
  add_test(NAME embedded_contract COMMAND jshc_embedded_tests)
endif()

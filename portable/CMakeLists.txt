cmake_minimum_required(VERSION 3.20)
project(jshc_portable VERSION 1.0.0 LANGUAGES C)
option(BUILD_SHARED_LIBS "Build shared library instead of static library" OFF)
add_library(jshc_portable src/jshc_portable.c)
target_include_directories(jshc_portable PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(jshc_portable PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(jshc_portable PUBLIC JSHC_P_SHARED PRIVATE JSHC_P_BUILD)
endif()
if(MSVC)
  target_compile_options(jshc_portable PRIVATE /W4)
else()
  target_compile_options(jshc_portable PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
include(GNUInstallDirs)
install(TARGETS jshc_portable ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/jshc_portable.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

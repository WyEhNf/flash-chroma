cmake_minimum_required(VERSION 3.24)
project(flash_chroma_single_header LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FLASH_CHROMA_TEST_OPENMP "验证单头文件的可选OpenMP调度" ON)
option(FLASH_CHROMA_TEST_AVX2 "验证单头文件的运行时AVX2分派" ON)
include(CTest)
get_filename_component(FC_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_library(flash_chroma_bridge SHARED bridge.cpp c_bridge.c)
target_include_directories(flash_chroma_bridge PRIVATE "${FC_ROOT}/include")
add_custom_command(TARGET flash_chroma_bridge POST_BUILD
  COMMAND "${CMAKE_COMMAND}" "-DFC_LIBRARY=$<TARGET_FILE:flash_chroma_bridge>"
    "-DFC_ROOT=${FC_ROOT}" "-DFC_COMPILER=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}"
    -P "${CMAKE_CURRENT_LIST_DIR}/build_receipt.cmake")
if(FLASH_CHROMA_TEST_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  target_link_libraries(flash_chroma_bridge PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(flash_chroma_bridge PRIVATE FLASH_CHROMA_ENABLE_OPENMP=1)
else()
  target_compile_definitions(flash_chroma_bridge PRIVATE FLASH_CHROMA_ENABLE_OPENMP=0)
endif()
if(NOT FLASH_CHROMA_TEST_AVX2)
  target_compile_definitions(flash_chroma_bridge PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(flash_chroma_bridge PRIVATE -Wall -Wextra -Wpedantic -fstack-usage)
endif()
if(BUILD_TESTING)
  add_executable(fc_c_contract "${FC_ROOT}/tests/single_header_c_contract.c")
  target_include_directories(fc_c_contract PRIVATE "${FC_ROOT}/include")
  add_test(NAME single_header_c_contract COMMAND fc_c_contract)
  add_executable(fc_runtime_contract "${FC_ROOT}/tests/single_header_runtime_contract.cpp")
  target_include_directories(fc_runtime_contract PRIVATE "${FC_ROOT}/include")
  if(FLASH_CHROMA_TEST_OPENMP)
    target_link_libraries(fc_runtime_contract PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(fc_runtime_contract PRIVATE FLASH_CHROMA_ENABLE_OPENMP=1)
  endif()
  if(NOT FLASH_CHROMA_TEST_AVX2)
    target_compile_definitions(fc_runtime_contract PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
  endif()
  add_test(NAME single_header_runtime_contract COMMAND fc_runtime_contract)
  foreach(contract direct prepare prepare_off)
    if(contract STREQUAL "prepare_off")
      set(contract_source prepare)
    else()
      set(contract_source ${contract})
    endif()
    add_executable(fc_${contract}_contract "${FC_ROOT}/tests/single_header_${contract_source}_contract.cpp")
    target_include_directories(fc_${contract}_contract PRIVATE "${FC_ROOT}/include")
    if(FLASH_CHROMA_TEST_OPENMP)
      target_link_libraries(fc_${contract}_contract PRIVATE OpenMP::OpenMP_CXX)
      target_compile_definitions(fc_${contract}_contract PRIVATE FLASH_CHROMA_ENABLE_OPENMP=1)
    endif()
    if(NOT FLASH_CHROMA_TEST_AVX2)
      target_compile_definitions(fc_${contract}_contract PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
    endif()
    if(contract STREQUAL "prepare_off")
      target_compile_definitions(fc_${contract}_contract PRIVATE FLASH_CHROMA_ENABLE_PREPARE_CACHE=0)
    endif()
    add_test(NAME single_header_${contract}_contract COMMAND fc_${contract}_contract)
  endforeach()
  add_executable(fc_selection_contract "${FC_ROOT}/tests/single_header_selection_contract.cpp")
  target_include_directories(fc_selection_contract PRIVATE "${FC_ROOT}/include")
  if(MSVC)
    target_compile_options(fc_selection_contract PRIVATE /UNDEBUG)
  else()
    target_compile_options(fc_selection_contract PRIVATE -UNDEBUG)
  endif()
  add_test(NAME single_header_selection_contract COMMAND fc_selection_contract)
  add_executable(fc_odr_contract odr_impl.cpp odr_user.cpp)
  target_include_directories(fc_odr_contract PRIVATE "${FC_ROOT}/include")
  add_test(NAME single_header_multiple_translation_units COMMAND fc_odr_contract)
  foreach(deployment RANGE 1 4)
    foreach(c_first RANGE 0 1)
      set(deployment_target "fc_deployment_${deployment}_${c_first}")
      add_executable(${deployment_target} "${FC_ROOT}/tests/single_header_deployment_contract.cpp")
      target_include_directories(${deployment_target} PRIVATE "${FC_ROOT}/include")
      target_compile_definitions(${deployment_target} PRIVATE FLASH_CHROMA_TARGET=${deployment} FC_DEPLOYMENT_C_FIRST=${c_first})
      if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${deployment_target} PRIVATE -fno-exceptions -fno-rtti)
      endif()
      add_test(NAME ${deployment_target} COMMAND ${deployment_target})
    endforeach()
  endforeach()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(deployment RANGE 3 4)
      add_library(fc_freestanding_${deployment} OBJECT "${FC_ROOT}/tests/single_header_cpp_freestanding.cpp")
      target_include_directories(fc_freestanding_${deployment} PRIVATE "${FC_ROOT}/include")
      target_compile_definitions(fc_freestanding_${deployment} PRIVATE FLASH_CHROMA_TARGET=${deployment})
      target_compile_options(fc_freestanding_${deployment} PRIVATE -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-exceptions -fno-rtti)
    endforeach()
  endif()
endif()

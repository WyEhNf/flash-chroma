cmake_minimum_required(VERSION 3.24)
project(flash_chroma_single_header LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FLASH_CHROMA_TEST_OPENMP "验证单头文件的可选OpenMP调度" ON)
option(FLASH_CHROMA_TEST_AVX2 "验证单头文件的运行时AVX2分派" ON)
include(CTest)
get_filename_component(FC_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_library(flash_chroma_bridge SHARED bridge.cpp c_bridge.c)
target_include_directories(flash_chroma_bridge PRIVATE "${FC_ROOT}/include")
add_executable(flash_chroma_run event_run.cpp)
target_include_directories(flash_chroma_run PRIVATE "${FC_ROOT}/include")
add_custom_command(TARGET flash_chroma_bridge POST_BUILD
  COMMAND "${CMAKE_COMMAND}" "-DFC_LIBRARY=$<TARGET_FILE:flash_chroma_bridge>"
    "-DFC_ROOT=${FC_ROOT}" "-DFC_COMPILER=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}"
    -P "${CMAKE_CURRENT_LIST_DIR}/build_receipt.cmake")
if(FLASH_CHROMA_TEST_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  target_link_libraries(flash_chroma_bridge PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(flash_chroma_bridge PRIVATE FLASH_CHROMA_ENABLE_OPENMP=1)
  target_link_libraries(flash_chroma_run PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(flash_chroma_run PRIVATE FLASH_CHROMA_ENABLE_OPENMP=1)
else()
  target_compile_definitions(flash_chroma_bridge PRIVATE FLASH_CHROMA_ENABLE_OPENMP=0)
  target_compile_definitions(flash_chroma_run PRIVATE FLASH_CHROMA_ENABLE_OPENMP=0)
endif()
if(NOT FLASH_CHROMA_TEST_AVX2)
  target_compile_definitions(flash_chroma_bridge PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
  target_compile_definitions(flash_chroma_run PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(flash_chroma_bridge PRIVATE -Wall -Wextra -Wpedantic -fstack-usage)
  target_compile_options(flash_chroma_run PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  # 比赛适配合同始终单线程；同时编译运行时分派与禁AVX2回退，沿用现有CI的CTest入口。
  foreach(framework_variant adapter adapter_scalar)
    set(framework_target "fc_framework_${framework_variant}_contract")
    add_executable(${framework_target} "${FC_ROOT}/tests/framework_adapter_contract.cpp")
    target_include_directories(${framework_target} PRIVATE "${FC_ROOT}/framework" "${FC_ROOT}/include")
    target_compile_definitions(${framework_target} PRIVATE FLASH_CHROMA_ENABLE_OPENMP=0)
    if(framework_variant STREQUAL "adapter_scalar" OR NOT FLASH_CHROMA_TEST_AVX2)
      target_compile_definitions(${framework_target} PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(${framework_target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME framework_${framework_variant}_contract COMMAND ${framework_target})
    set_tests_properties(framework_${framework_variant}_contract PROPERTIES LABELS "framework;single-thread" TIMEOUT 180 PROCESSORS 1)
  endforeach()
  # 拟合接口与小型数值合同不运行性能计时，统一进入现有CTest/CI门禁。
  foreach(fit_contract fit fit_numerics)
    set(fit_target "fc_framework_${fit_contract}_contract")
    add_executable(${fit_target} "${FC_ROOT}/tests/framework_${fit_contract}_contract.cpp")
    target_include_directories(${fit_target} PRIVATE "${FC_ROOT}/framework" "${FC_ROOT}/include")
    target_compile_definitions(${fit_target} PRIVATE FLASH_CHROMA_ENABLE_OPENMP=0)
    if(NOT FLASH_CHROMA_TEST_AVX2)
      target_compile_definitions(${fit_target} PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(${fit_target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME framework_${fit_contract}_contract COMMAND ${fit_target})
    set_tests_properties(framework_${fit_contract}_contract PROPERTIES LABELS "framework;fit;single-thread" TIMEOUT 180 PROCESSORS 1)
  endforeach()
  # 连续偏好使用合成指标和小型Stage，不生成稠密表、不运行校准计时。
  add_executable(fc_framework_preference_contract "${FC_ROOT}/tests/framework_preference_contract.cpp")
  target_include_directories(fc_framework_preference_contract PRIVATE "${FC_ROOT}/framework" "${FC_ROOT}/include")
  target_compile_definitions(fc_framework_preference_contract PRIVATE FLASH_CHROMA_ENABLE_OPENMP=0)
  if(NOT FLASH_CHROMA_TEST_AVX2)
    target_compile_definitions(fc_framework_preference_contract PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(fc_framework_preference_contract PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME framework_preference_contract COMMAND fc_framework_preference_contract)
  set_tests_properties(framework_preference_contract PROPERTIES LABELS "framework;preference;single-thread" TIMEOUT 30 PROCESSORS 1)
  add_executable(fc_c_contract "${FC_ROOT}/tests/single_header_c_contract.c")
  target_include_directories(fc_c_contract PRIVATE "${FC_ROOT}/include")
  add_test(NAME single_header_c_contract COMMAND fc_c_contract)
  add_executable(fc_runtime_contract "${FC_ROOT}/tests/single_header_runtime_contract.cpp")
  target_include_directories(fc_runtime_contract PRIVATE "${FC_ROOT}/include")
  if(FLASH_CHROMA_TEST_OPENMP)
    target_link_libraries(fc_runtime_contract PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(fc_runtime_contract PRIVATE FLASH_CHROMA_ENABLE_OPENMP=1)
  endif()
  if(NOT FLASH_CHROMA_TEST_AVX2)
    target_compile_definitions(fc_runtime_contract PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
  endif()
  add_test(NAME single_header_runtime_contract COMMAND fc_runtime_contract)
  add_executable(fc_rgb24_contract "${FC_ROOT}/tests/single_header_rgb24_contract.cpp")
  target_include_directories(fc_rgb24_contract PRIVATE "${FC_ROOT}/include")
  target_compile_definitions(fc_rgb24_contract PRIVATE FLASH_CHROMA_ENABLE_OPENMP=0)
  if(NOT FLASH_CHROMA_TEST_AVX2)
    target_compile_definitions(fc_rgb24_contract PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(fc_rgb24_contract PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|i.86)$" OR
       CMAKE_CXX_COMPILER_ARCHITECTURE_ID MATCHES "^(x64|X64|x86_64|X86)$")
      target_compile_options(fc_rgb24_contract PRIVATE -mno-bmi -mno-bmi2)
    endif()
  endif()
  add_test(NAME single_header_rgb24_contract COMMAND fc_rgb24_contract)
  set_tests_properties(single_header_rgb24_contract PROPERTIES
    LABELS "direct;rgb24;full-domain;single-thread" TIMEOUT 180 PROCESSORS 1 SKIP_RETURN_CODE 77)
  foreach(contract direct prepare prepare_off)
    if(contract STREQUAL "prepare_off")
      set(contract_source prepare)
    else()
      set(contract_source ${contract})
    endif()
    add_executable(fc_${contract}_contract "${FC_ROOT}/tests/single_header_${contract_source}_contract.cpp")
    target_include_directories(fc_${contract}_contract PRIVATE "${FC_ROOT}/include")
    if(FLASH_CHROMA_TEST_OPENMP)
      target_link_libraries(fc_${contract}_contract PRIVATE OpenMP::OpenMP_CXX)
      target_compile_definitions(fc_${contract}_contract PRIVATE FLASH_CHROMA_ENABLE_OPENMP=1)
    endif()
    if(NOT FLASH_CHROMA_TEST_AVX2)
      target_compile_definitions(fc_${contract}_contract PRIVATE FLASH_CHROMA_ENABLE_AVX2=0)
    endif()
    if(contract STREQUAL "prepare_off")
      target_compile_definitions(fc_${contract}_contract PRIVATE FLASH_CHROMA_ENABLE_PREPARE_CACHE=0)
    endif()
    add_test(NAME single_header_${contract}_contract COMMAND fc_${contract}_contract)
  endforeach()
  add_executable(fc_selection_contract "${FC_ROOT}/tests/single_header_selection_contract.cpp")
  target_include_directories(fc_selection_contract PRIVATE "${FC_ROOT}/include")
  if(MSVC)
    target_compile_options(fc_selection_contract PRIVATE /UNDEBUG)
  else()
    target_compile_options(fc_selection_contract PRIVATE -UNDEBUG)
  endif()
  add_test(NAME single_header_selection_contract COMMAND fc_selection_contract)
  add_executable(fc_odr_contract odr_impl.cpp odr_user.cpp)
  target_include_directories(fc_odr_contract PRIVATE "${FC_ROOT}/include")
  add_test(NAME single_header_multiple_translation_units COMMAND fc_odr_contract)
  foreach(deployment RANGE 1 4)
    foreach(c_first RANGE 0 1)
      set(deployment_target "fc_deployment_${deployment}_${c_first}")
      add_executable(${deployment_target} "${FC_ROOT}/tests/single_header_deployment_contract.cpp")
      target_include_directories(${deployment_target} PRIVATE "${FC_ROOT}/include")
      target_compile_definitions(${deployment_target} PRIVATE FLASH_CHROMA_TARGET=${deployment} FC_DEPLOYMENT_C_FIRST=${c_first})
      if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${deployment_target} PRIVATE -fno-exceptions -fno-rtti)
      endif()
      add_test(NAME ${deployment_target} COMMAND ${deployment_target})
    endforeach()
  endforeach()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(deployment RANGE 3 4)
      add_library(fc_freestanding_${deployment} OBJECT "${FC_ROOT}/tests/single_header_cpp_freestanding.cpp")
      target_include_directories(fc_freestanding_${deployment} PRIVATE "${FC_ROOT}/include")
      target_compile_definitions(fc_freestanding_${deployment} PRIVATE FLASH_CHROMA_TARGET=${deployment})
      target_compile_options(fc_freestanding_${deployment} PRIVATE -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-exceptions -fno-rtti)
    endforeach()
  endif()
endif()
